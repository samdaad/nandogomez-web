name: CI
on:
  pull_request:
  push:
    branches: [main]
permissions:
  contents: read
concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: true
jobs:
  verify:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    env:
      PUBLIC_GA_ID: ${{ vars.PUBLIC_GA_ID }}
    steps:
      - uses: actions/checkout@v5
      - uses: actions/setup-node@v5
        with:
          node-version-file: .node-version
          cache: npm
      - run: npm ci
      - name: Typecheck
        run: npm run typecheck
      - name: Tests de unidad
        run: npm test
      - name: Build
        run: npm run build
      - name: Tests sobre dist
        run: npm run --if-present test:dist
      - name: E2E (Playwright)
        if: hashFiles('playwright.config.ts') != ''
        run: |
          npx playwright install --with-deps chromium
          npm run test:e2e
      - name: Lint (informativo)
        continue-on-error: true
        run: npm run --if-present lint
      - name: Pendientes de publicación (informativo)
        continue-on-error: true
        run: npm run --if-present check:release
